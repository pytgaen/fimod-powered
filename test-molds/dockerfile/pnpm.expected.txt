FROM node:22-slim

RUN mkdir -p /tmp \
 && chmod 1777 /tmp \
 && groupadd -r node \
 && useradd -r -g node -d /app -s /sbin/nologin node

WORKDIR /app

RUN chown node:node /app

RUN corepack enable
COPY --chown=node:node package.json pnpm-lock.yaml* ./
RUN pnpm install --frozen-lockfile --prod

COPY --chown=node:node . .

EXPOSE 3000

USER node

CMD ["node","index.js"]
